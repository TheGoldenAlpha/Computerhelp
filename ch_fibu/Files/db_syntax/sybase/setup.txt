CREATE TABLE IF NOT EXISTS kontenrahmen_vfibu (kontonr VARCHAR(10) DEFAULT NULL, name VARCHAR(200) NOT NULL, id UNIQUEIDENTIFIER NOT NULL, hierarchie INTEGER DEFAULT NULL, parent_konto UNIQUEIDENTIFIER DEFAULT NULL, sort_columns INTEGER DEFAULT NULL, mandanten_id UNIQUEIDENTIFIER DEFAULT NULL, year INTEGER DEFAULT NULL, saldo NUMERIC(8,2), PRIMARY KEY (id));
CREATE TABLE IF NOT EXISTS buchungen_vfibu (buchungsid UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID(), "date" DATE NOT NULL, konto1 VARCHAR(37) NOT NULL, konto2 VARCHAR(37), buchungstext VARCHAR(150), Betrag DECIMAL(10,2), steuercode VARCHAR(5), kostenstelle VARCHAR(100), Jahr INTEGER, Mandantenid VARCHAR(37), PRIMARY KEY (buchungsid));
